def handler(event, context):
    # Временно проверяем, что именно передаёт API Gateway
    path_params = event.get('pathParams') or {}

    text = path_params.get('text')

    if text is None:
        return {
            'statusCode': 400,
            'headers': {'Content-Type': 'application/json'},
            'body': __import__('json').dumps(event, ensure_ascii=False, default=str)
        }

    if not text or any(c < 'a' or c > 'z' for c in text):
        return {
            'statusCode': 400,
            'headers': {'Content-Type': 'text/plain; charset=utf-8'},
            'body': 'Invalid input'
        }

    return {
        'statusCode': 200,
        'headers': {'Content-Type': 'text/plain; charset=utf-8'},
        'body': text[::-1]
    }
